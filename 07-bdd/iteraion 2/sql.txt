CREATE TABLE Citoyen(
   User_id VARCHAR(50),
   User_email VARCHAR(50) NOT NULL,
   User_mdp VARCHAR(50) NOT NULL,
   User_name VARCHAR(50) NOT NULL,
   PRIMARY KEY(User_id),
   UNIQUE(User_email)
);

CREATE TABLE Categorie(
   Categorie_id VARCHAR(50),
   Categorie_nom VARCHAR(50) NOT NULL,
   PRIMARY KEY(Categorie_id),
   UNIQUE(Categorie_nom)
);

CREATE TABLE Zone(
   Zone_id VARCHAR(50),
   Zone_nom VARCHAR(50),
   PRIMARY KEY(Zone_id)
);

CREATE TABLE Lieu(
   Lieu_id VARCHAR(50),
   Lieu_latitude DECIMAL(9,6) NOT NULL,
   Lieu_longtitude DECIMAL(9,6) NOT NULL,
   Zone_id VARCHAR(50),
   Categorie_id VARCHAR(50) NOT NULL,
   PRIMARY KEY(Lieu_id),
   FOREIGN KEY(Zone_id) REFERENCES Zone(Zone_id),
   FOREIGN KEY(Categorie_id) REFERENCES Categorie(Categorie_id)
);

CREATE TABLE Demande(
   Demande_id VARCHAR(50),
   Demande_type VARCHAR(20) NOT NULL,
   Demande_description TEXT,
   Demande_date TIMESTAMP NOT NULL,
   Lieu_id VARCHAR(50) NOT NULL,
   User_id VARCHAR(50) NOT NULL,
   PRIMARY KEY(Demande_id),
   FOREIGN KEY(Lieu_id) REFERENCES Lieu(Lieu_id),
   FOREIGN KEY(User_id) REFERENCES Citoyen(User_id)
);

SELECT 
    D.Demande_id, 
    D.Demande_type, 
    D.Demande_description, 
    D.Demande_date,
    C.User_name AS Citoyen,
    C.User_email,
    L.Lieu_id,
    L.Lieu_latitude,
    L.Lieu_longtitude
FROM Demande D
JOIN Citoyen C ON D.User_id = C.User_id
JOIN Lieu L ON D.Lieu_id = L.Lieu_id
ORDER BY D.Demande_date DESC;
